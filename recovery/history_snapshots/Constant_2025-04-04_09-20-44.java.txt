	public static Color gradient(Color color1, Color color2, double grad) {
		int r = (int)(color1.getRed() * (1 - grad) + color2.getRed() * grad)%256;
		int g = (int)(color1.getGreen() * (1 - grad) + color2.getGreen() * grad)%256;
		int b = (int)(color1.getBlue() * (1 - grad) + color2.getBlue() * grad)%256;
		int a = (int) Constant.border((color1.getAlpha() * (1 - grad) + color2.getAlpha() * grad)%256,255,0);
		return(new Color(r, g, b, a));
	}
	public static int mod(int a, int b) {
